pipeline{
    
    agent any
    
    environment{
        image_name = "pycloud5858/python-app"
        tag = """${BUILD_NUMBER}"""
    }
    stages{
        
        stage("GitHub Checkout"){
            steps{
                git branch: 'main', credentialsId: 'GitHub-PAT', url: 'https://github.com/pycloudnative/python-app.git'
            }
        }
        
        stage("Docker Image Build"){
            steps{
                sh "docker build -t ${image_name}:${tag} ."
            }
        }
        
        stage("Docker Push"){
            steps{
                withDockerRegistry(credentialsId: 'Docker-Hub', url: 'https://index.docker.io/v1/') {
            sh "docker push ${image_name}:${tag}"
}
            }
        }
        
        stage("K8s YAML Creation"){
            steps{
                fileOperations([fileCreateOperation(fileContent: """apiVersion: apps/v1
kind: Deployment
metadata:
  name: python-app-deployment
  labels:
    app: python-app
spec:
  selector:
    matchLabels:
      app: python-app
  template:
    metadata:
      labels:
        app: python-app
    spec:
      containers:
      - name: python-app
        image: ${image_name}:${tag}
        ports:
        - containerPort: 80
---
apiVersion: v1
kind: Service
metadata:
  name: python-app-service
spec:
  selector:
    app: python-app
  ports:
    - protocol: TCP
      port: 80
  type: NodePort


""", fileName: '/var/lib/jenkins/workspace/python-app-dev/deploy.yml')])
            }
        }
        
        stage ("K8s Deployment") {
            steps {
               script {
            kubernetesDeploy(configs: "deploy.yml", kubeconfigId: "dev-k8s-cluster")
        }
            }
        }

        
    }
}
